# Embed the existing Fcitx pinyin engine, without installing another IME.
function(tailkana_add_libime)
  set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
  set(BUILD_SHARED_LIBS OFF)
  set(ENABLE_TEST OFF)
  set(ENABLE_DATA OFF)
  set(ENABLE_TOOLS ON)
  add_subdirectory("${TAILKANA_ROOT}/work/libime" "${CMAKE_BINARY_DIR}/tailkana-libime")
endfunction()

# The installed AI host is native and self-contained; no Python/server runtime.
function(tailkana_add_ai_service)
  set(BUILD_SHARED_LIBS OFF)
  set(LLAMA_BUILD_TESTS OFF)
  set(LLAMA_BUILD_EXAMPLES OFF)
  set(LLAMA_BUILD_SERVER OFF)
  set(LLAMA_CURL OFF)
  set(GGML_METAL_EMBED_LIBRARY ON)
  add_subdirectory("${TAILKANA_ROOT}/work/ai-experiment/llama.cpp" "${CMAKE_BINARY_DIR}/tailkana-llama" EXCLUDE_FROM_ALL)
endfunction()
tailkana_add_ai_service()
foreach(ai_library llama ggml ggml-base ggml-cpu ggml-metal)
  if(TARGET ${ai_library})
    target_compile_options(${ai_library} PRIVATE -O3)
  endif()
endforeach()
add_executable(TailKanaAI ai_service.cpp mozc_backend.cpp "${TAILKANA_MOZC}/src/unix/fcitx5/mozc_direct_client.cc")

target_link_libraries(TailKanaAI PRIVATE llama Fcitx5::Core mozc-static libprotobuf absl::flags_parse absl::log_flags)
