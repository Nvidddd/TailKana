cmake_minimum_required(VERSION 3.22)
project(TailKanaWindowsCore VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(tailkana-windows-core STATIC core/session.cpp ../src/roman_buffer.cpp ../src/roman_correction.cpp ../src/typing_policy.cpp)
target_include_directories(tailkana-windows-core PUBLIC core ../src)
if(MSVC)
  target_compile_options(tailkana-windows-core PUBLIC /utf-8 /W4)
else()
  target_compile_options(tailkana-windows-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(tailkana-windows-session-test tests/session_test.cpp)
target_link_libraries(tailkana-windows-session-test PRIVATE tailkana-windows-core)
add_test(NAME windows-session COMMAND tailkana-windows-session-test)
add_executable(tailkana-windows-roman-test ../tests/roman_buffer_test.cpp)
target_link_libraries(tailkana-windows-roman-test PRIVATE tailkana-windows-core)
add_test(NAME windows-roman COMMAND tailkana-windows-roman-test)
if(WIN32)
  add_library(tailkana-windows-panel STATIC native/panel.cpp)
  target_include_directories(tailkana-windows-panel PUBLIC native)
  target_compile_definitions(tailkana-windows-panel PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(tailkana-windows-panel PUBLIC tailkana-windows-core user32 gdi32)
  add_executable(tailkana-windows-panel-test tests/panel_test.cpp)
  target_link_libraries(tailkana-windows-panel-test PRIVATE tailkana-windows-panel)
  add_test(NAME windows-panel COMMAND tailkana-windows-panel-test)
endif()
if(WIN32)
  add_executable(tailkana-windows-registration-test tests/registration_test.cpp)
  target_compile_definitions(tailkana-windows-registration-test PRIVATE UNICODE _UNICODE)
  target_link_libraries(tailkana-windows-registration-test PRIVATE ole32 oleaut32 uuid msi)
endif()
